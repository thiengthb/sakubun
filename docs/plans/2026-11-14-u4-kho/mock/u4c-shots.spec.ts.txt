// Scratch (never committed): screenshots of the three U4c layouts at /resources/zz-u4c.
import path from 'node:path';
import Database from 'better-sqlite3';
import { expect, test } from '@playwright/test';
import { E2E_DB_FILE, E2E_EMAIL, E2E_LOGIN, E2E_OWNER, seedE2eAccount } from './fixtures';

const OUT = process.env.SHOT_DIR!;
const DAY = 86_400_000;

async function open(browser: import('@playwright/test').Browser, w: number) {
  const ctx = await browser.newContext({
    viewport: { width: w, height: w === 390 ? 844 : 900 },
    deviceScaleFactor: 2,
  });
  const page = await ctx.newPage();
  await page.addInitScript(() => {
    const s = document.createElement('style');
    s.textContent = 'nextjs-portal{display:none!important}';
    document.addEventListener('DOMContentLoaded', () => document.head.append(s));
  });
  await page.goto('/login');
  await page.getByLabel('Email').fill(E2E_EMAIL);
  await page.getByLabel('Mật khẩu').fill(E2E_LOGIN);
  await page.getByRole('button', { name: 'Đăng nhập' }).click();
  await page.waitForURL((u) => !u.pathname.startsWith('/login'), { timeout: 30_000 });
  return { ctx, page };
}

test.beforeAll(seedE2eAccount);

test('u4c layouts', async ({ browser }) => {
  test.setTimeout(600_000);
  const file = path.join(process.cwd(), E2E_DB_FILE);
  {
    // Take N5 from Hôm nay, as a new learner does.
    const { ctx, page } = await open(browser, 1440);
    const d = new Database(file);
    for (const t of ['GroupMember', 'Group', 'ReviewLog', 'SrsState', 'Sentence', 'Item', 'Settings'])
      d.exec(`DELETE FROM "${t}"`);
    d.close();
    await page.goto('/', { waitUntil: 'networkidle' });
    await page.getByRole('region', { name: 'Hôm nay' }).getByRole('button', { name: 'N5' }).click();
    await expect(page.getByRole('link', { name: 'Bắt đầu ôn' })).toBeVisible({ timeout: 90_000 });
    await ctx.close();
  }
  const d = new Database(file);
  const popular = d
    .prepare(
      `SELECT expression FROM LibraryEntry WHERE type='VOCAB' AND level='N5' AND expression IS NOT NULL
       ORDER BY origin DESC, takeCount DESC, createdAt ASC, id ASC LIMIT 60`,
    )
    .all() as { expression: string }[];
  // Some words the learner does not have: a few near the top, and a run further down.
  const drop = new Set([3, 8, 12, 17, ...Array.from({ length: 20 }, (_, i) => 40 + i)]);
  const now = Date.now();
  const plan: [number, number][] = [
    [2, -DAY], [2, 12 * DAY], [1, 2 * DAY], [0, 0], [2, 20 * DAY], [1, -DAY], [1, 3 * DAY],
    [0, 0], [2, 9 * DAY], [2, 30 * DAY], [1, DAY], [0, 0], [2, -DAY], [2, 15 * DAY],
  ];
  let k = 0;
  popular.forEach(({ expression }, i) => {
    const item = d
      .prepare(`SELECT id FROM Item WHERE userId = ${E2E_OWNER} AND kind='VOCAB' AND expression = ?`)
      .get(expression) as { id: string } | undefined;
    if (!item) return;
    if (drop.has(i)) {
      d.prepare(`DELETE FROM SrsState WHERE itemId = ?`).run(item.id);
      d.prepare(`DELETE FROM Item WHERE id = ?`).run(item.id);
      return;
    }
    const [state, off] = plan[k++ % plan.length];
    d.prepare(`UPDATE SrsState SET state = ?, due = ?, reps = ? WHERE itemId = ?`).run(
      state,
      new Date(now + off).toISOString(),
      state ? 3 : 0,
      item.id,
    );
  });
  d.close();
  for (const w of [1440, 390]) {
    const { ctx, page } = await open(browser, w);
    for (const v of ['a', 'b', 'c']) {
      await page.goto(`/resources/zz-u4c?v=${v}`, { waitUntil: 'networkidle' });
      await page.waitForTimeout(800);
      await page.screenshot({ path: `${OUT}/${v}-${w}.png` });
    }
    await ctx.close();
  }
});
