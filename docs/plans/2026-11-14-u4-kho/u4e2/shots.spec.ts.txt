import path from 'node:path';

import { E2E_DB_FILE, E2E_EMAIL, E2E_LOGIN, E2E_OWNER, seedE2eAccount } from './fixtures';

const OUT = process.env.SHOT_DIR!;
const DAY = 86_400_000;

async function open(browser: import('@playwright/test').Browser, w: number) {
  const ctx = await browser.newContext({ viewport: { width: w, height: w === 390 ? 844 : 900 }, deviceScaleFactor: 2 });
  const page = await ctx.newPage();
  await page.addInitScript(() => {
    const s = document.createElement('style');
    s.textContent = 'nextjs-portal{display:none!important}';
    document.addEventListener('DOMContentLoaded', () => document.head.append(s));
  });
  await page.goto('/login');
  await page.getByLabel('Email').fill(E2E_EMAIL);
  await page.getByLabel('Mật khẩu').fill(E2E_LOGIN);
  await page.getByRole('button', { name: 'Đăng nhập' }).click();
  await page.waitForURL((u) => !u.pathname.startsWith('/login'), { timeout: 30_000 });
  return { ctx, page };
}

const GROUPS: [string, number[]][] = [
  ['Từ trong bài dịch tuần này', [0, 1, 4, 5, 7, 8]],
  ['Công sở', [10, 11, 12, 13, 14, 6, 9]],
];

function prepare() {
  const d = new Database(path.join(process.cwd(), E2E_DB_FILE));
  d.pragma('foreign_keys = ON');
  for (const t of ['GroupMember', 'Group', 'ReviewLog', 'SrsState', 'Sentence', 'Item']) d.exec(`DELETE FROM "${t}"`);
  const owner = (d.prepare(`SELECT ${E2E_OWNER} AS id`).get() as { id: string }).id;
  const now = Date.now();
  ITEMS.forEach(([expr, reading, meaning, source, st], i) => {
    const created = new Date(now - (40 - i) * DAY).toISOString();
    d.prepare(
      `INSERT INTO Item (id, userId, kind, expression, reading, meaningVi, level, source, suspended, starred, createdAt)
       VALUES (?, ?, 'VOCAB', ?, ?, ?, 'N5', ?, 0, 0, ?)`,
    ).run(`u4e${i}`, owner, expr, reading, meaning, source, created);
    const state = st === 'n' ? 0 : st === 'l' ? 1 : 2;
    const due = st === 'd' ? now - DAY : st === 'n' ? now : now + 9 * DAY;
    d.prepare(`INSERT INTO SrsState (id, userId, itemId, due, state, reps) VALUES (?, ?, ?, ?, ?, ?)`).run(
      `s${i}`, owner, `u4e${i}`, new Date(due).toISOString(), state, state === 0 ? 0 : 3,
    );
  });
  GROUPS.forEach(([name, members], g) => {
    const at = new Date(now - g * DAY).toISOString();
    d.prepare(`INSERT INTO "Group" (id, userId, name, kind, createdAt, updatedAt) VALUES (?, ?, ?, 'mixed', ?, ?)`).run(`g${g}`, owner, name, at, at);
    for (const m of members) d.prepare(`INSERT INTO GroupMember (groupId, itemId, userId, addedAt) VALUES (?, ?, ?, ?)`).run(`g${g}`, `u4e${m}`, owner, at);
  });
  d.close();
}

test('u4e2 shots', async ({ browser }) => {
  test.setTimeout(600_000);
  prepare();
  for (const w of [1440, 390]) {
    const { ctx, page } = await open(browser, w);
    const shot = async (name: string) => {
      await page.waitForTimeout(700);
      await page.screenshot({ path: `${OUT}/${name}-${w}.jpg`, type: 'jpeg', quality: 85 });
    };
    await page.goto('/resources/outside', { waitUntil: 'networkidle' });
    await page.getByRole('button', { name: 'Thêm' }).click();
    const panel = page.getByRole('dialog', { name: 'Thêm vào kho' });
    await shot('sheet');
    await panel.getByLabel(/Từ \(tiếng Nhật\)/).fill('段取り八分');
    await panel.getByLabel(/Cách đọc/).fill('だんどりはちぶ');
    await panel.getByLabel(/Nghĩa/).fill('chuẩn bị tốt là xong tám phần');
    await panel.getByRole('button', { name: 'Thêm từ' }).scrollIntoViewIfNeeded();
    await shot('manual');
    await ctx.close();
  }
});
