// Scratch (never committed as .ts): KD2 review shots. Copy into e2e/ as kd2-shots.spec.ts, SHOT_DIR=<dir>.
import path from 'node:path';
import { readFileSync } from 'node:fs';
import { test, expect, type Page } from '@playwright/test';
import { pathPoints, STROKE_BOX } from '@/lib/japanese/strokes';
import { E2E_EMAIL, E2E_LOGIN, seedE2eAccount } from './fixtures';

const OUT = process.env.SHOT_DIR!;
const TABLE = JSON.parse(
  readFileSync(path.join(process.cwd(), 'data/kanji/strokes.json'), 'utf8'),
) as Record<string, string[]>;

// Drawn a little loosely, as a hand would, so the guesses are not just the one exact match.
async function draw(page: Page, kanji: string, upTo = Infinity) {
  const pad = page.locator('[data-slot="kanji-draw"]');
  const box = (await pad.boundingBox())!;
  let seed = 3;
  const jitter = () => ((seed = (seed * 16807) % 2147483647) / 2147483647 - 0.5) * 3;
  for (const [n, d] of TABLE[kanji].entries()) {
    if (n >= upTo) break;
    const dx = jitter();
    const dy = jitter();
    const pts = pathPoints(d).map((p) => ({
      x: box.x + ((p.x * 0.9 + 6 + dx + jitter() * 0.3) / STROKE_BOX) * box.width,
      y: box.y + ((p.y * 0.9 + 4 + dy + jitter() * 0.3) / STROKE_BOX) * box.height,
    }));
    await page.mouse.move(pts[0].x, pts[0].y);
    await page.mouse.down();
    for (const p of pts.slice(1)) await page.mouse.move(p.x, p.y, { steps: 2 });
    await page.mouse.up();
    await expect(pad).toHaveAttribute('data-strokes', String(n + 1));
  }
  await page.waitForTimeout(800);
}

test.beforeAll(seedE2eAccount);

test('kd2 shots', async ({ browser }) => {
  test.setTimeout(600_000);
  for (const w of [1440, 390]) {
    const ctx = await browser.newContext({
      viewport: { width: w, height: w === 390 ? 844 : 900 },
      deviceScaleFactor: 2,
      hasTouch: w === 390,
    });
    const page = await ctx.newPage();
    await page.addInitScript(() => {
      const s = document.createElement('style');
      s.textContent = 'nextjs-portal{display:none!important}';
      document.addEventListener('DOMContentLoaded', () => document.head.append(s));
    });
    await page.goto('/login');
    await page.getByLabel('Email').fill(E2E_EMAIL);
    await page.getByLabel('Mật khẩu').fill(E2E_LOGIN);
    await page.getByRole('button', { name: 'Đăng nhập' }).click();
    await page.waitForURL((u) => !u.pathname.startsWith('/login'), { timeout: 30_000 });
    await page.goto('/resources', { waitUntil: 'networkidle' });
    await page.waitForTimeout(500);
    const shot = (name: string) =>
      page.screenshot({ path: `${OUT}/${name}-${w}.jpg`, type: 'jpeg', quality: 85 });
    await shot('closed');
    await page.getByRole('button', { name: 'Vẽ chữ kanji' }).click();
    await page.waitForTimeout(400);
    await shot('open');
    await draw(page, '社');
    await shot('drawing');
    await page.getByRole('button', { name: 'Thêm 社' }).click();
    await page.waitForURL(/q=/);
    await page.waitForLoadState('networkidle');
    await draw(page, '会', 4);
    await shot('second');
    await ctx.close();
  }
});
